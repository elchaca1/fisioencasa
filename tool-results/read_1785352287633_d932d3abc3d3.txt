    83→/* ─────────── Scroll Animation Wrapper ─────────── */
    84→function AnimatedSection({
    85→  children,
    86→  className = '',
    87→  delay = 0,
    88→}: {
    89→  children: React.ReactNode
    90→  className?: string
    91→  delay?: number
    92→}) {
    93→  const ref = useRef<HTMLDivElement>(null)
    94→  const isInView = useInView(ref, { once: true, margin: '-80px' })
    95→
    96→  return (
    97→    <motion.div
    98→      ref={ref}
    99→      initial={{ opacity: 0, y: 40 }}
   100→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 40 }}
   101→      transition={{ duration: 0.6, delay, ease: 'easeOut' }}
   102→      className={className}
   103→    >
   104→      {children}
   105→    </motion.div>
   106→  )
   107→}
